const G = 9.81; const u = 80; // Python-style round(): half-to-even on exact ties function round(x, n = 0) { const s = x.toFixed(Math.min(100, n + 30)); const dot = s.indexOf("."); const tail = s.slice(dot + 1 + n); if (/^50*$/.test(tail)) { const t = s.slice(0, n === 0 ? dot : dot + 1 + n); if (Number(t[t.length - 1]) % 2 === 0) return Number(t); } return Number(x.toFixed(n)); } // Python prints floats like 5.0 rather than 5 const pf = (v) => (Number.isInteger(v) ? v.toFixed(1) : String(v)); async function get_float(prompt) { while (true) { const ans = (await ask(prompt)).trim(); if (ans === "") return 0.0; const num = Number(ans); if (!Number.isNaN(num)) return num; console.log("Enter a valid number: "); } } function check(d, s) { if (u ** 4 - G * (G * d ** 2 + 2 * s * u ** 2) >= 0) return true; else return false; } function time(theta, d) { return d / (u * Math.cos((theta * Math.PI) / 180)); } async function find_elevation(dx, dy) { const range = await get_float("Enter range from your range finder: "); const horizontalD = (dx ** 2 + dy ** 2) ** (1 / 2); return (range ** 2 - horizontalD ** 2) ** (1 / 2); } function calculation(dx, dy, s) { const d = (dx ** 2 + dy ** 2) ** (1 / 2); if (check(d, s)) { let k1, k2; if (dx === 0 && dy === 0) { k1 = 0; k2 = 0; } else { k1 = (u ** 2 - (u ** 4 - G * (G * d ** 2 + 2 * s * u ** 2)) ** 0.5) / (G * d); k2 = (u ** 2 + (u ** 4 - G * (G * d ** 2 + 2 * s * u ** 2)) ** 0.5) / (G * d); } const direct_angle = (Math.atan(k1) * 180) / Math.PI; const indirect_angle = (Math.atan(k2) * 180) / Math.PI; let bearing; try { bearing = (Math.atan2(dx, dy) * 180) / Math.PI; if (bearing < 0) bearing += 360; } catch (e) { bearing = 0; } const text = ` Distance from target: ${pf(round(d, 2))} Direct fire angle: ${pf(round(direct_angle, 1))}° Direction: ${Math.trunc(round(bearing - (1 * time(direct_angle, d)) / 12, 0))}° Time of travel: ${pf(round(time(direct_angle, d), 1))} seconds Indirect fire angle: ${pf(round(indirect_angle, 1))}° Time of travel: ${pf(round(time(indirect_angle, d), 1))} seconds Direction: ${Math.trunc(round(bearing - (1 * time(indirect_angle, d)) / 12, 0))}° id: ${dx},${dy},${s} `;
